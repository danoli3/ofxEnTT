cmake_minimum_required(VERSION 3.22)
project(ofxentt_revision_comparison LANGUAGES C CXX)

set(OF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." CACHE PATH "openFrameworks root")
set(OFXENTT_BEFORE_DIR "" CACHE PATH "Extracted baseline addon sources")
set(OFXENTT_AFTER_DIR "" CACHE PATH "Extracted merged addon sources")
option(OFXENTT_CORE_BUNDLES_FREEIMAGE "Local prebuilt core already contains FreeImage" OFF)
include("${OF_ROOT}/libs/openFrameworksCompiled/project/CMake/openFrameworks.cmake")
of_use_core()
if(OFXENTT_CORE_BUNDLES_FREEIMAGE)
    get_target_property(core_links openFrameworks_deps INTERFACE_LINK_LIBRARIES)
    list(FILTER core_links EXCLUDE REGEX "/FreeImage[.]a$")
    set_target_properties(openFrameworks_deps PROPERTIES INTERFACE_LINK_LIBRARIES "${core_links}")
endif()

foreach(revision before merged)
    if(revision STREQUAL "before")
        set(addon_dir "${OFXENTT_BEFORE_DIR}")
    else()
        set(addon_dir "${OFXENTT_AFTER_DIR}")
    endif()
    if(NOT EXISTS "${addon_dir}/src/ofxEnTTWorld.cpp")
        message(FATAL_ERROR "Missing ${revision} snapshot. Run run.py to prepare and build both revisions.")
    endif()
    file(GLOB addon_sources CONFIGURE_DEPENDS "${addon_dir}/src/*.cpp")
    add_executable(${revision}_bench compare.cpp ${addon_sources})
    target_include_directories(${revision}_bench PRIVATE "${addon_dir}/src" "${addon_dir}/libs/entt/include")
    target_link_libraries(${revision}_bench PRIVATE openFrameworks::openFrameworks)
    if(revision STREQUAL "merged")
        target_compile_definitions(${revision}_bench PRIVATE OFXENTT_MERGED=1)
    endif()
endforeach()
add_executable(compare_states validate.cpp)
foreach(target before_bench merged_bench compare_states)
    set_target_properties(${target} PROPERTIES
        INTERPROCEDURAL_OPTIMIZATION FALSE
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    foreach(config Debug Release RelWithDebInfo MinSizeRel)
        string(TOUPPER "${config}" upper)
        set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/bin")
    endforeach()
endforeach()
