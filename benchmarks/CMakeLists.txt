cmake_minimum_required(VERSION 3.22)
project(ofxentt_benchmarks LANGUAGES C CXX)
set(OF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." CACHE PATH "openFrameworks root")
option(OFXENTT_CORE_BUNDLES_FREEIMAGE "Local prebuilt core already contains FreeImage" OFF)
include("${OF_ROOT}/libs/openFrameworksCompiled/project/CMake/openFrameworks.cmake")
of_add_app(ofxentt_benchmarks ADDONS ofxEnTT)
if(OFXENTT_CORE_BUNDLES_FREEIMAGE)
    get_target_property(core_links openFrameworks_deps INTERFACE_LINK_LIBRARIES)
    list(FILTER core_links EXCLUDE REGEX "/FreeImage[.]a$")
    set_target_properties(openFrameworks_deps PROPERTIES INTERFACE_LINK_LIBRARIES "${core_links}")
endif()
